\begin{enumerate}

\end{enumerate}

\begin{align*}
m_R(\vec{x}, 0) & = 0\\
m_R(\vec{x}, y+1) & =
\begin{cases}
m_R(\vec{x}, y) & \text{if $m_R(\vec x, y) \neq y$}\\
y & \text{if $m_R(\vec x, y) = y$ and $R(\vec{x}, y)$}\\
y+1 & \text{otherwise.}
\end{cases}
\end{align*}
Note that there is a $z<y$ such that $R(\vec x, z)$ iff $m_R(\vec x,
y) \neq y$.
\end{proof}
