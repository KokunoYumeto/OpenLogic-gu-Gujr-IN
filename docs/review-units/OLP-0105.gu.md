\begin{defn}[!!^{derivability}]

\begin{defn}[!!^{derivability}]

!!{sentence}s~$\Gamma$, $\Gamma \Proves !A$ iff there is a
finite set $\{!B_1, \dots, !B_n\} \subseteq \Gamma$
and a closed !!{tableau} for the set

\{

  \sFmla{\True}{!B_1}, \dots,

  \}.

!A$.
\end{defn}
