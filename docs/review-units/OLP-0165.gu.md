\begin{prop}[Extensionality]

  !!{structure}s with $\Domain{M_1} = \Domain{M_2}$, and $s$ a

  $\Sat{M_1}{!A}[s]$ iff $\Sat{M_2}{!A}[s]$.
\end{prop}
